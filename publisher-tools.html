function runSEOCheck() {
  let missingFAQ = 0;
  let missingAuthor = 0;
  let missingRelated = 0;
  let missingDateModified = 0;
  let duplicates = 0;

  let urls = new Set();

  allArticles.forEach(a => {
    if (!a.faq || !a.faq.length) missingFAQ++;
    if (!a.author) missingAuthor++;
    if (!a.relatedArticles || !a.relatedArticles.length) missingRelated++;
    if (!a.dateModified) missingDateModified++;

    if (urls.has(a.url)) duplicates++;
    urls.add(a.url);
  });

  document.getElementById("seo").innerText =
`Missing FAQ: ${missingFAQ}
Missing Author: ${missingAuthor}
Missing dateModified: ${missingDateModified}
Missing Related: ${missingRelated}
Duplicate URLs: ${duplicates}`;
}

function buildOverview() {
  let skincare = allArticles.filter(a => a.category === "Skincare").length;
  let haircare = allArticles.filter(a => a.category === "Haircare").length;
  let comparison = allArticles.filter(a => (a.url || "").includes("vs")).length;

  document.getElementById("overview").innerText =
`Total Articles: ${allArticles.length}
Skincare: ${skincare}
Haircare: ${haircare}
Comparison: ${comparison}`;
}

function updateHealth() {
  let seo = document.getElementById("seo").innerText;

  let issues = seo.match(/: [1-9][0-9]*/g);
  let totalIssues = issues ? issues.length : 0;

  let health = "🟢 Excellent";

  if (totalIssues >= 1 && totalIssues <= 3) health = "🟡 Good";
  else if (totalIssues <= 6) health = "🟠 Needs Attention";
  else if (totalIssues > 6) health = "🔴 Errors Found";

  document.getElementById("health").innerText =
    "Website Health: " + health;
}

function generateSitemap() {
  let xml = `\n\n`;

  allArticles.forEach(a => {
    xml += `

  https://dailyglamstore.in${a.url}
`;
  });

  xml += `\n`;

  document.getElementById("output").innerText = xml;
}

function copySitemap() {
  navigator.clipboard.writeText(document.getElementById("output").innerText);
  alert("Sitemap copied!");
}
